// machine-readable code for clients, and a human-readable message.

type AppError struct {
        Status  int
        Code    string
        Message string
}

func (e AppError) Error() string { return e.Code + ": " + e.Message }

// Response converts the error into the standard API envelope.
func (e AppError) Response() server.Response {
        return server.ErrorResponse(e.Code, e.Message)
}

func New(status int, code, message string) AppError {
        return AppError{Status: status, Code: code, Message: message}
}

// Request errors: the request itself is malformed or unroutable (400 / 404 / 405 / 413).
var (
        EmptyRequestBody = New(http.StatusBadRequest, "empty_request_body", "request body is required")
        InvalidJSON      = New(http.StatusBadRequest, "invalid_json", "request body must be a single valid JSON object")
        RequestTooLarge  = New(http.StatusRequestEntityTooLarge, "request_too_large", "request body is too large")
        RouteNotFound    = New(http.StatusNotFound, "route_not_found", "no endpoint matches this path")
        MethodNotAllowed = New(http.StatusMethodNotAllowed, "method_not_allowed", "this endpoint does not support the request method")
)

// Calculation errors: the request is well-formed but the math is not
// defined for these inputs (422 Unprocessable Entity).
var (
        DivisionByZero    = New(http.StatusUnprocessableEntity, "division_by_zero", "division by zero is undefined")
        UndefinedResult   = New(http.StatusUnprocessableEntity, "undefined_result", "result is not a real number")
        ResultOutOfRange  = New(http.StatusUnprocessableEntity, "result_out_of_range", "result is too large to represent")
        InternalServerErr = New(http.StatusInternalServerError, "internal_server_error", "internal server error")
)

func MissingOperand(field string) AppError {
        return New(http.StatusBadRequest, "missing_operand", fmt.Sprintf("field %q is required", field))
}

func InvalidOperand(field string) AppError {
        return New(http.StatusBadRequest, "invalid_operand", fmt.Sprintf("field %q must be a finite number", field))
}

func UnknownField(detail string) AppError {
        return New(http.StatusBadRequest, "unknown_field", detail)
}

// FromDomain maps service-layer errors to API errors. Unknown errors become
// a generic 500 so internal details never leak to clients.
func FromDomain(err error) AppError {
        switch {
        case errors.Is(err, service.ErrDivisionByZero):
                return DivisionByZero
        case errors.Is(err, service.ErrUndefinedResult):
                return UndefinedResult
        case errors.Is(err, service.ErrResultOutOfRange):
                return ResultOutOfRange
        case errors.Is(err, service.ErrInvalidOperand):
                return New(http.StatusBadRequest, "invalid_operand", "operands must be finite numbers")
        default:
                return InternalServerErr
        }
}

import (

// CalculatorServicePort is the contract the HTTP layer depends on.
// Depending on an interface (rather than the concrete type) keeps handlers
// easy to test with a fake implementation.
type CalculatorServicePort interface {
        Add(a, b float64) (float64, error)
        Subtract(a, b float64) (float64, error)
        Multiply(a, b float64) (float64, error)
        Divide(a, b float64) (float64, error)
        Power(base, exponent float64) (float64, error)
        Sqrt(a float64) (float64, error)
        Percentage(a, b float64) (float64, error)
}

// CalculatorService implements the arithmetic operations.
// It is stateless; every method is a pure function of its inputs.
type CalculatorService struct{}

// Compile-time check that CalculatorService satisfies the port.
var _ CalculatorServicePort = (*CalculatorService)(nil)

func NewCalculatorService() *CalculatorService {
        return &CalculatorService{}
}

func (s *CalculatorService) Add(a, b float64) (float64, error) {
        return compute(func() float64 { return a + b }, a, b)
}

func (s *CalculatorService) Subtract(a, b float64) (float64, error) {
        return compute(func() float64 { return a - b }, a, b)
}

func (s *CalculatorService) Multiply(a, b float64) (float64, error) {
        return compute(func() float64 { return a * b }, a, b)
}

func (s *CalculatorService) Divide(a, b float64) (float64, error) {
        if err := validateOperands(a, b); err != nil {
                return 0, err
        }
        if b == 0 {
                return 0, ErrDivisionByZero
        }
        return compute(func() float64 { return a / b }, a, b)
}

func (s *CalculatorService) Power(base, exponent float64) (float64, error) {
        if err := validateOperands(base, exponent); err != nil {
                return 0, err
        }
        // 0 ^ -n == 1 / 0^n, which is a division by zero.
        if base == 0 && exponent < 0 {
                return 0, ErrDivisionByZero
        }
        return compute(func() float64 { return math.Pow(base, exponent) }, base, exponent)
}

func (s *CalculatorService) Sqrt(a float64) (float64, error) {
        if err := validateOperands(a); err != nil {
                return 0, err
        }
        if a < 0 {
                return 0, ErrUndefinedResult
        }
        return compute(func() float64 { return math.Sqrt(a) }, a)
}

// Percentage returns b percent of a, i.e. a * b / 100.
// Example: Percentage(200, 10) == 20.
//
// Dividing first (a / 100 * b) avoids an intermediate overflow such as
// 1e308 * 50 overflowing before the division would bring it back in range.
func (s *CalculatorService) Percentage(a, b float64) (float64, error) {
        return compute(func() float64 { return a / 100 * b }, a, b)
}

// compute validates the operands, runs op and normalizes its result.
func compute(op func() float64, operands ...float64) (float64, error) {
        if err := validateOperands(operands...); err != nil {
                return 0, err
        }
        return normalize(op())
}

func validateOperands(operands ...float64) error {
        for _, v := range operands {
                if math.IsNaN(v) || math.IsInf(v, 0) {
                        return ErrInvalidOperand
                }
        }
        return nil
}

// significantDigits is the precision results are rounded to.
// float64 reliably represents 15 significant decimal digits, so rounding there
// removes binary floating-point noise (0.1 + 0.2 = 0.30000000000000004 -> 0.3)
// without losing any meaningful precision.
const significantDigits = 15

// normalize turns NaN/Inf into domain errors and rounds away float noise.
func normalize(result float64) (float64, error) {
        switch {
        case math.IsNaN(result):
                return 0, ErrUndefinedResult
        case math.IsInf(result, 0):
                return 0, ErrResultOutOfRange
        }

        rounded, err := strconv.ParseFloat(strconv.FormatFloat(result, 'g', significantDigits, 64), 64)
        if err != nil {
                // Rounding a value next to math.MaxFloat64 up can overflow;
                // the unrounded value is still a valid, finite result.
                rounded = result
        }

        // Avoid returning "-0" (e.g. -5 * 0) to clients.
        if rounded == 0 {
                return 0, nil
        }
        return rounded, nil
}

        h.handleBinary(w, r, "divide", h.s.Divide)
}

func (h *CalculatorHandler) Power(w http.ResponseWriter, r *http.Request) {
        h.handleBinary(w, r, "power", h.s.Power)
}

        h.handleBinary(w, r, "percentage", h.s.Percentage)
}

func (h *CalculatorHandler) Sqrt(w http.ResponseWriter, r *http.Request) {
        var req dto.UnaryOperationRequest
        if appErr := decodeJSON(w, r, &req); appErr != nil {
                writeError(w, *appErr)
                return
        }
        if req.A == nil {
                writeError(w, apperror.MissingOperand("a"))
                return
        }

        result, err := h.s.Sqrt(*req.A)
        writeResult(w, "sqrt", result, err)
}

// handleBinary holds the flow shared by every two-operand endpoint:
// decode -> validate -> calculate -> respond.
func (h *CalculatorHandler) handleBinary(
        w http.ResponseWriter,
        r *http.Request,
        operation string,
        calculate func(a, b float64) (float64, error),
) {
        var req dto.BinaryOperationRequest
        if appErr := decodeJSON(w, r, &req); appErr != nil {
                writeError(w, *appErr)
                return
        }
        if req.A == nil {
                writeError(w, apperror.MissingOperand("a"))
                return
        }
        if req.B == nil {
                writeError(w, apperror.MissingOperand("b"))
                return
        }

        result, err := calculate(*req.A, *req.B)
        writeResult(w, operation, result, err)
}

func writeResult(w http.ResponseWriter, operation string, result float64, err error) {
        if err != nil {
                writeError(w, apperror.FromDomain(err))
                return
        }
        server.WriteJSON(w, http.StatusOK, server.SuccessResponse(dto.OperationResponse{
                Operation: operation,
                Result:    result,
        }))
}

func writeError(w http.ResponseWriter, e apperror.AppError) {
        server.WriteJSON(w, e.Status, e.Response())
}

import (

        "errors"
        "io"
        "net/http"
        "strings"

        "github.com/gghinzu/calculator/backend/internal/apperror"
)

// maxBodyBytes caps request bodies. A valid request is a few dozen bytes,
// so 1 KiB leaves plenty of room while rejecting abusive payloads early.
const maxBodyBytes = 1 << 10

// decodeJSON strictly decodes a single JSON object into dst and translates
// every decoding failure into a specific, client-friendly API error.
func decodeJSON(w http.ResponseWriter, r *http.Request, dst any) *apperror.AppError {
        r.Body = http.MaxBytesReader(w, r.Body, maxBodyBytes)

        dec := json.NewDecoder(r.Body)
        dec.DisallowUnknownFields()

        if err := dec.Decode(dst); err != nil {
                appErr := translateDecodeError(err)
                return &appErr
        }

        // Reject trailing data such as `{"a":1}{"a":2}` or `{"a":1} garbage`.
        if err := dec.Decode(&struct{}{}); !errors.Is(err, io.EOF) {
                var maxBytesErr *http.MaxBytesError
                if errors.As(err, &maxBytesErr) {
                        return &apperror.RequestTooLarge
                }
                return &apperror.InvalidJSON
        }
        return nil
}

func translateDecodeError(err error) apperror.AppError {
        var (
                typeErr     *json.UnmarshalTypeError
                maxBytesErr *http.MaxBytesError
        )

        switch {
        case errors.Is(err, io.EOF):
                return apperror.EmptyRequestBody
        case errors.As(err, &maxBytesErr):
                return apperror.RequestTooLarge
        case errors.As(err, &typeErr) && typeErr.Field != "":
                // Covers wrong types ({"a": "5"}) and numbers outside the float64
                // range ({"a": 1e400}).
                return apperror.InvalidOperand(typeErr.Field)
        case strings.HasPrefix(err.Error(), "json: unknown field "):
                // encoding/json has no exported type for this error.
                return apperror.UnknownField(strings.TrimPrefix(err.Error(), "json: "))
        default:
                // Syntax errors, truncated bodies, a top-level array, etc.
                return apperror.InvalidJSON
        }
}

import (

// Middleware wraps an http.Handler with extra behaviour.
type Middleware func(http.Handler) http.Handler

// Chain applies middlewares so that the first one is the outermost:
// Chain(h, A, B) == A(B(h)).
func Chain(h http.Handler, mws ...Middleware) http.Handler {
        for i := len(mws) - 1; i >= 0; i-- {
                h = mws[i](h)
        }
        return h
}

// Recover turns a panic in any handler into a JSON 500 instead of dropping
// the connection.
func Recover(next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                defer func() {
                        if rec := recover(); rec != nil {
                                slog.Error("panic recovered", "panic", rec, "path", r.URL.Path)
                                e := apperror.InternalServerErr
                                server.WriteJSON(w, e.Status, e.Response())
                        }
                }()
                next.ServeHTTP(w, r)
        })
}

// Logger logs one structured line per request.
func Logger(next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                start := time.Now()
                rec := &statusRecorder{ResponseWriter: w, status: http.StatusOK}
                next.ServeHTTP(rec, r)
                slog.Info("request",
                        "method", r.Method,
                        "path", r.URL.Path,
                        "status", rec.status,
                        "duration", time.Since(start),
                )
        })
}

type statusRecorder struct {
        http.ResponseWriter
        status int
}

func (r *statusRecorder) WriteHeader(status int) {
        r.status = status
        r.ResponseWriter.WriteHeader(status)
}

// CORS allows browser requests from allowedOrigin. An empty origin disables
// CORS entirely, which is the default: in development the Vite dev server
// proxies /api, and in Docker the frontend is served from the same origin.
func CORS(allowedOrigin string) Middleware {
        return func(next http.Handler) http.Handler {
                if allowedOrigin == "" {
                        return next
                }
                return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                        h := w.Header()
                        h.Set("Access-Control-Allow-Origin", allowedOrigin)
                        h.Set("Access-Control-Allow-Methods", "GET, POST, OPTIONS")
                        h.Set("Access-Control-Allow-Headers", "Content-Type")
                        h.Add("Vary", "Origin")

                        if r.Method == http.MethodOptions {
                                w.WriteHeader(http.StatusNoContent)
                                return
                        }
                        next.ServeHTTP(w, r)
                })
        }
}

// jsonRouteErrors makes the mux's own 404 / 405 responses use the same JSON
// envelope as every other API error, instead of net/http's plain-text body.
//
// ServeMux.Handler reports an empty pattern when no route matched. In that
// case the mux's response is captured; 404 and 405 are rewritten as JSON
// (keeping the Allow header), anything else — e.g. a 301 path-cleanup
// redirect — is passed through unchanged.
func jsonRouteErrors(mux *http.ServeMux) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                if _, pattern := mux.Handler(r); pattern != "" {
                        mux.ServeHTTP(w, r)
                        return
                }

                captured := newCaptureWriter()
                mux.ServeHTTP(captured, r)

                switch captured.status {
                case http.StatusNotFound:
                        writeError(w, apperror.RouteNotFound)
                case http.StatusMethodNotAllowed:
                        w.Header().Set("Allow", captured.Header().Get("Allow"))
                        writeError(w, apperror.MethodNotAllowed)
                default:
                        captured.replay(w)
                }
        })
}

func writeError(w http.ResponseWriter, e apperror.AppError) {
        server.WriteJSON(w, e.Status, e.Response())
}

// captureWriter buffers a response so it can be inspected before sending.
type captureWriter struct {
        header http.Header
        status int
        body   bytes.Buffer
}

func newCaptureWriter() *captureWriter {
        return &captureWriter{header: http.Header{}, status: http.StatusOK}
}

func (c *captureWriter) Header() http.Header         { return c.header }
func (c *captureWriter) Write(b []byte) (int, error) { return c.body.Write(b) }
func (c *captureWriter) WriteHeader(status int)      { c.status = status }

func (c *captureWriter) replay(w http.ResponseWriter) {
        for k, v := range c.header {
                w.Header()[k] = v
        }
        w.WriteHeader(c.status)
        _, _ = w.Write(c.body.Bytes())
}

import (

// Options configures the router.
type Options struct {
        // CORSOrigin enables CORS for one origin; empty disables it.
        CORSOrigin string
        // StaticDir, when set, serves the built frontend from that directory, so
        // a single process (and a single Docker container) serves the whole app.
        StaticDir string
}

// New wires every route and the middleware stack.
// It uses the standard library's method-aware routing (Go 1.22+), so wrong
// methods get a 405 and unknown paths a 404 without any framework; both are
// returned in the standard JSON error envelope.
func New(h *handler.CalculatorHandler, opts Options) http.Handler {
        mux := http.NewServeMux()

        mux.HandleFunc("GET /health", handler.Health)

        mux.HandleFunc("POST /api/v1/add", h.Add)
        mux.HandleFunc("POST /api/v1/subtract", h.Subtract)
        mux.HandleFunc("POST /api/v1/multiply", h.Multiply)
        mux.HandleFunc("POST /api/v1/divide", h.Divide)
        mux.HandleFunc("POST /api/v1/power", h.Power)
        mux.HandleFunc("POST /api/v1/sqrt", h.Sqrt)
        mux.HandleFunc("POST /api/v1/percentage", h.Percentage)

        var root http.Handler = jsonRouteErrors(mux)
        if opts.StaticDir != "" {
                root = withStatic(root, opts.StaticDir)
        }

        return middleware.Chain(root,
                middleware.Recover,
                middleware.Logger,
                middleware.CORS(opts.CORSOrigin),
        )
}

// withStatic sends API paths to the API and everything else to the static
// frontend. Keeping the two apart means an unknown /api path is still a JSON
// 404 and never falls through to an HTML page.
func withStatic(api http.Handler, dir string) http.Handler {
        files := http.FileServer(http.Dir(dir))
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                if isAPIPath(r.URL.Path) {
                        api.ServeHTTP(w, r)
                        return
                }
                // Vite puts a content hash in every asset file name, so those files
                // never change and can be cached for good. index.html is not cached,
                // so a new deploy is picked up immediately.
                if strings.HasPrefix(r.URL.Path, "/assets/") {
                        w.Header().Set("Cache-Control", "public, max-age=31536000, immutable")
                } else {
                        w.Header().Set("Cache-Control", "no-cache")
                }
                files.ServeHTTP(w, r)
        })
}

func isAPIPath(path string) bool {
        return path == "/health" || path == "/api" || strings.HasPrefix(path, "/api/")
}

import (

        "log/slog"

// Response is the envelope every API response uses, so clients can always
// check `success` first and then read either `data` or `error`.
type Response struct {
        Success bool   `json:"success"`
        Data    any    `json:"data,omitempty"`
        Error   *Error `json:"error,omitempty"`
}

type Error struct {
        Code    string `json:"code"`
        Message string `json:"message"`
}

func SuccessResponse(data any) Response {
        return Response{Success: true, Data: data}
}

func ErrorResponse(code, message string) Response {
        return Response{Success: false, Error: &Error{Code: code, Message: message}}
}

// WriteJSON serializes body as JSON with the given status code.
func WriteJSON(w http.ResponseWriter, status int, body any) {
        w.Header().Set("Content-Type", "application/json")
        w.WriteHeader(status)
        if err := json.NewEncoder(w).Encode(body); err != nil {
                // Headers are already sent, so the best we can do is log it.
                slog.Error("failed to encode response", "error", err)
        }
}

package config

import "os"

// Config holds runtime settings read from environment variables.
// A few plain env vars do not justify a config library (the earlier project
// used viper), so this stays on the standard library.
type Config struct {
        // Port the HTTP server listens on. Env: PORT (default 8080).
        Port string
        // CORSAllowedOrigin enables CORS for one origin when set.
        // Env: CORS_ALLOWED_ORIGIN (default: disabled).
        CORSAllowedOrigin string
        // StaticDir serves the built frontend from this directory when set.
        // Env: STATIC_DIR (default: API only). Used by the Docker image.
        StaticDir string
}

func Load() Config {
        return Config{
                Port:              getEnv("PORT", "8080"),
                CORSAllowedOrigin: os.Getenv("CORS_ALLOWED_ORIGIN"),
                StaticDir:         os.Getenv("STATIC_DIR"),
        }
}

func getEnv(key, fallback string) string {
        if v, ok := os.LookupEnv(key); ok && v != "" {
                return v
        }
        return fallback
}
